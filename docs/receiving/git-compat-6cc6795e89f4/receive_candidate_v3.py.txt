def normalized_files(module, text):
    return [dict(path=x.path,old_path=x.old_path,is_new=x.is_new,is_deleted=x.is_deleted,
                 added_lines=sorted(x.added_lines),touched_lines=sorted(x.touched_lines))
            for x in module.parse_unified_diff(text)]
def normalized_functions(module, text):
    return [dict(path=x.path,qualname=x.qualname,lineno=x.lineno,end_lineno=x.end_lineno,
                 changed_lines=x.changed_lines,source=x.source,is_method=x.is_method)
            for x in module.changed_functions(REPO,text)]
def observe(module, text):
    try:
        files = normalized_files(module,text)
        functions = normalized_functions(module,text)
        return {"ok":True,"files":files,"functions":functions,
                "targets":sorted([[x["path"],x["qualname"]] for x in functions])}
    except Exception as exc:
        return {"ok":False,"error_type":type(exc).__name__,"message":str(exc),"traceback":traceback.format_exc()}

    return {"argv":[sys.executable,"-B","-m","testpilot",*map(str,argv)],"returncode":p.returncode,
            "stdout":p.stdout.decode("utf-8","backslashreplace"),"stderr":p.stderr.decode("utf-8","backslashreplace")}
