def run(argv, cwd, ok=(0,)):
    p = subprocess.run(argv, cwd=cwd, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
    commands.append({"argv":list(map(str, argv)), "cwd":str(cwd), "returncode":p.returncode,
                     "stdout":p.stdout.decode("utf-8", "backslashreplace"),
                     "stderr":p.stderr.decode("utf-8", "backslashreplace")})
    if p.returncode not in ok:
        raise RuntimeError(commands[-1])
    return p.stdout

def add_case(name, patch, expected, zero=False, optional=False, require=()):
    for marker in require:
        assert marker in patch, (name, marker)
    path = PATCHES / (name + ".diff")
    path.write_bytes(patch)
    argv = ["git", "-c", "core.autocrlf=false", "apply", "--check"]
    if zero:
        argv.append("--unidiff-zero")

    run(argv, ORACLE)
    cases.append({"name":name, "patch":str(path), "sha256":hashlib.sha256(patch).hexdigest(),
                  "expected_targets":sorted(expected), "git_valid":True, "optional_envelope":optional})
